    const timeZone = Intl.DateTimeFormat().resolvedOptions().timeZone;

      const currentDate = new Date();

<div id="searchF">	
    <input type="text" id="searchInput" placeholder="Enter text to search...">
    <button id="searchButton">Search</button>
</div>

<script>
    let searchT = "";

	document.getElementById("icon").onclick = function() {
		if ( document.getElementById("searchF").style.display === "none"){
			document.getElementById("searchF").style.display = "block";
		}else{
			document.getElementById("searchF").style.display = "none";
		}
	};

    // 搜尋按鈕點擊事件
    document.getElementById('searchButton').addEventListener('click', function() {
        const searchText = document.getElementById('searchInput').value.trim();
        if (searchText) {
            searchT = searchText;
            performSearch(searchT, false); // 開始新的搜尋
        }
    });

    // 監聽 Enter 鍵找下一個
    document.addEventListener('keydown', function(event) {
        if (event.key === 'Enter') {
            const searchText = document.getElementById('searchInput').value.trim();
            if (searchText) {
                // 如果搜尋文字變了，重新開始；否則找下一個
                if (searchText !== searchT) {
                    searchT = searchText;
                    performSearch(searchT, false);
                } else {
                    performSearch(searchT, true);
                }
                event.preventDefault(); // 防止表單提交或換行
            }
        }
    });

    /**
     * 核心搜尋函數
     * @param {string} text - 要搜尋的文字
     * @param {boolean} next - 是否搜尋下一個
     */
    function performSearch(text, next) {
        // 使用 window.find 是最簡單且支援「找下一個」的瀏覽器原生方法
        // 參數說明: (字串, 是否區分大小寫, 是否向後搜尋, 是否循環搜尋, 是否全字匹配, 是否搜尋框, 是否向前搜尋)
        const found = window.find(text, false, false, true, false, true, false);

        if (found) {
            // 取得目前選中的區域並確保它在視線內
            const selection = window.getSelection();
            if (selection.rangeCount > 0) {
                const range = selection.getRangeAt(0);
                const element = range.startContainer.parentElement;
                
                // 視覺回饋：讓找到的文字閃爍或顯示
                element.scrollIntoView({ behavior: 'smooth', block: 'center' });
            }
        } else {
            // 如果到底了，再跑一次會從頭開始（因為上面設置了循環搜尋 true）
            // 或者你可以給予提示
            alert('已到達文件末尾，將從頭開始搜尋。');
        }
    }
</script>
